Storage Rules:

rules_version = '2'; service firebase.storage { match /b/{bucket}/o { // Drivers folder - users can upload to their own folder match /drivers/{userId}/{fileName} { // Allow read: authenticated users can read their own files allow read: if request.auth != null && request.auth.uid == userId; // Allow write: authenticated users can write to their own folder // with validation for file type (images only) and size (max 10MB) allow write: if request.auth != null && request.auth.uid == userId && request.resource.contentType.matches('image/.*') && request.resource.size < 10 * 1024 * 1024; } // Default: deny all other access match /{allPaths=**} { allow read, write: if false; } } }
